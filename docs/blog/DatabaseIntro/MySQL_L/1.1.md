# MySQL 基本操作
## 1.1 DDL

```sql
# 查看所有数据库
show databases;

# 查询当前数据库
select database();

# 创建数据库
create database if not exists test;

# 删除数据库
drop database if exists test;

# 切换数据库
use test;

# 查看所有表
show tables;

# 创建表
create table if not exists test(
    id int,
    name varchar(20)
) comment '测试表';

# 删除表
drop table if exists test;
```
